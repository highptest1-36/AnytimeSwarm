% \panel{x1}{y1}{x2}{y2}{colour}{tab text}{tab width}
\newcommand{\panel}[7]{%
  \draw[#5, line width=1.1pt, rounded corners=6pt, fill=white] (#1,#2) rectangle (#3,#4);
  \node[H, fill=#5, rounded corners=4pt, minimum height=0.5cm, minimum width=#7cm, anchor=west]
    at ({#1+0.25},#4) {#6};
}
% \sub{x1}{y1}{x2}{y2}{fill}{border}{number}{title}
\newcommand{\sub}[8]{%
  \filldraw[draw=#6, line width=0.5pt, rounded corners=5pt, fill=#5] (#1,#2) rectangle (#3,#4);
  \node[circle, draw=black!70, fill=white, line width=0.6pt, inner sep=0pt, minimum size=0.42cm,
        font=\fontsize{8}{8}\selectfont] at ({#1+0.33},{#4-0.32}) {#7};
  \node[S, anchor=west] at ({#1+0.6},{#4-0.32}) {#8};
}
% big chevron arrow: x, y (left tip of the shaft), length, colour
\newcommand{\chevron}[4]{%
  \fill[#4] (#1,{#2-0.12}) -- ++({#3-0.3},0) -- ++(0,-0.12) -- ++(0.3,0.24) -- ++(-0.3,0.24) -- ++(0,-0.12)
    -- ({#1},{#2+0.12}) -- cycle;}
% task pin: x, y, colour
\newcommand{\pin}[3]{%
  \filldraw[draw=black!60, line width=0.35pt, fill=#3] (#1,#2) -- ++(-0.09,0.17) arc (180:0:0.09) -- cycle;
  \fill[white] ({#1},{#2+0.19}) circle (0.035);}
