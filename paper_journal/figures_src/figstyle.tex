% \slab{x}{y}{width}{height}{depth}{colour}: isometric slab, front face at (x,y)
\newcommand{\slab}[6]{%
  \filldraw[draw=edge, line width=0.4pt, fill=#6!45!white]
    ({#1},{#2+#4}) -- ++({0.55*#5},{0.42*#5}) -- ++({#3},0) -- ++({-0.55*#5},{-0.42*#5}) -- cycle;
  \filldraw[draw=edge, line width=0.4pt, fill=#6!75!black]
    ({#1+#3},{#2}) -- ++({0.55*#5},{0.42*#5}) -- ++(0,{#4}) -- ++({-0.55*#5},{-0.42*#5}) -- cycle;
  \filldraw[draw=edge, line width=0.4pt, fill=#6] ({#1},{#2}) rectangle ++({#3},{#4});
}
% \wslab: white front face with a coloured top (software blocks that carry text)
\newcommand{\wslab}[6]{%
  \filldraw[draw=edge, line width=0.4pt, fill=#6!60!white]
    ({#1},{#2+#4}) -- ++({0.55*#5},{0.42*#5}) -- ++({#3},0) -- ++({-0.55*#5},{-0.42*#5}) -- cycle;
  \filldraw[draw=edge, line width=0.4pt, fill=#6!35!black!25]
    ({#1+#3},{#2}) -- ++({0.55*#5},{0.42*#5}) -- ++(0,{#4}) -- ++({-0.55*#5},{-0.42*#5}) -- cycle;
  \filldraw[draw=edge, line width=0.4pt, fill=white] ({#1},{#2}) rectangle ++({#3},{#4});
}
% \groundplane{x}{y}{width}{depth}{grid steps}: isometric ground with grid, front-left corner at (x,y)
\newcommand{\groundplane}[5]{%
  \filldraw[draw=muted, line width=0.4pt, fill=ground]
    ({#1},{#2}) -- ++({#3},0) -- ++({0.55*#4},{0.42*#4}) -- ++({-#3},0) -- cycle;
  \foreach \k in {1,...,#5} {
    \draw[black!15, line width=0.3pt] ({#1+\k*#3/(#5+1)},{#2}) -- ++({0.55*#4},{0.42*#4});
    \draw[black!15, line width=0.3pt] ({#1+0.55*#4*\k/(#5+1)},{#2+0.42*#4*\k/(#5+1)}) -- ++({#3},0);
  }
}

% task pillar (cylinder): x, y, height, colour
\newcommand{\pillar}[4]{%
  \filldraw[draw=edge, line width=0.3pt, fill=#4] ({#1-0.05},{#2}) -- ({#1-0.05},{#2+#3}) arc (180:0:0.05 and 0.02) -- ({#1+0.05},{#2}) arc (0:-180:0.05 and 0.02);
  \filldraw[draw=edge, line width=0.3pt, fill=#4!50!white] ({#1},{#2+#3}) ellipse (0.05 and 0.02);
}
